(function () {
  var q = document.getElementById('sk-q'), count = document.getElementById('sk-count');
  var entries = [].slice.call(document.querySelectorAll('.sk-entry'));
  var toc = [].slice.call(document.querySelectorAll('.sk-toc li'));
  function dedent(s) {
    var lines = s.replace(/^\n+|\s+$/g, '').split('\n');
    var min = Math.min.apply(null, lines.filter(function (l) { return l.trim(); }).map(function (l) { return l.match(/^ */)[0].length; }));
    return lines.map(function (l) { return l.slice(min); }).join('\n').replace(/=""/g, '');
  }
  q.addEventListener('input', function () {
    var v = q.value.trim().toLowerCase().split(/\s+/).filter(Boolean), n = 0;
    entries.forEach(function (e, i) {
      var ok = v.every(function (t) { return e.dataset.key.indexOf(t) > -1; });
      e.hidden = !ok; toc[i].hidden = !ok; if (ok) n++;
    });
    count.textContent = n + ' of ' + entries.length;
  });
  document.addEventListener('click', function (ev) {
    var a = ev.target.closest('.sk-entry__preview a[href="#"]');
    if (a) { ev.preventDefault(); return; }
    var b = ev.target.closest('[data-act]');
    if (!b) return;
    var en = b.closest('.sk-entry'), pv = en.querySelector('.sk-entry__preview'), code = en.querySelector('.sk-entry__code');
    var src = dedent(pv.innerHTML);
    if (b.dataset.act === 'code') { if (!code.textContent) code.textContent = src; code.hidden = !code.hidden; }
    else { navigator.clipboard.writeText(src).then(function () { var t = b.textContent; b.textContent = 'Copied'; setTimeout(function () { b.textContent = t; }, 1200); }); }
  });
})();
